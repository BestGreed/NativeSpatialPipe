cmake_minimum_required(VERSION 3.24)
project(NSP_AudioBackends LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "NSP native backends require Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_library(NSP.AsioDevice STATIC src/asio_backend.cpp src/asio_format.cpp src/diagnostics.cpp src/sinc_resampler.cpp)
target_include_directories(NSP.AsioDevice PUBLIC src PRIVATE third_party/asio/common)
target_compile_definitions(NSP.AsioDevice PRIVATE SPATIALASIO_VERSION="nsp-dev16")
target_link_libraries(NSP.AsioDevice PUBLIC ole32 advapi32 user32 uuid avrt winmm)
add_library(NSP.AsioOutput SHARED src/asio_output_c.cpp)
target_compile_definitions(NSP.AsioOutput PRIVATE NSP_ASIO_BUILD)
target_link_libraries(NSP.AsioOutput PRIVATE NSP.AsioDevice)
add_library(NSP.SpatialRenderer SHARED src/spatial_renderer_c.cpp src/native_scene.cpp src/metadata_renderer.cpp src/point_geometry.cpp src/speaker_identity.cpp)
target_compile_definitions(NSP.SpatialRenderer PRIVATE NSP_SPATIAL_BUILD)
target_include_directories(NSP.SpatialRenderer PUBLIC src)
# Compatibility helpers are not linked into either production DLL.
add_library(NSP.LegacyLayoutAdapter STATIC EXCLUDE_FROM_ALL src/asio_layout_adapter.cpp src/speaker_layout.cpp)
target_include_directories(NSP.LegacyLayoutAdapter PUBLIC src)
target_link_libraries(NSP.LegacyLayoutAdapter PUBLIC NSP.AsioDevice)
